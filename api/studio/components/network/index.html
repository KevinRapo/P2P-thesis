    network¶

    studio.components.network · source on GitHub

    Network building blocks shared by the Grid and Inspect pages.

    Each block builds its own controls and fills itself through the services; pages only place the blocks and connect them (e.g. "when the picker changes, load the summary"). Slow service calls run in a worker thread (run.io_bound) while the block shows a spinner.

    Classes:

    Name Description
    NetworkPicker

    Source toggle, voltage level and network dropdowns.
    ElementsTable

    The network's loads, generators and storage, optionally for one bus.
    ProfileInspector

    The network's SimBench profiles: a selectable table and two plots.

    Functions:

    Name Description
    busy

    Replace a box's content with a spinner while a slow call runs.
    failed

    Show an error in place of a box's content.
    yes_no

    yes / no (or the given text) for a flag; — when unknown.
    summary_facts

    The facts of a network as (label, value) rows, shared by summary and compare.
    show_summary

    Fill box with the network's name, counts and data flags.
    show_topology

    Draw the topology into box; a click on a bus calls on_click.
    clicked_bus

    The bus ID of a Plotly click event (buses carry it as customdata).
    profile_rows

    Table rows for profile types.

                {ALL: "All"}, value=ALL, label="Scenario"
            ).classes("grow basis-0")
            self.switches = ui.select(
                {ALL: "All", True: "With switches", False: "No switches"},
                value=ALL,
                label="Switches",
            ).classes("grow basis-0")
        self.facet_hint = ui.label().classes("mls-card-caption")

        ui.select({}, label="Network", with_input=True)
        .classes("w-full")
        .props("options-dense")

    self.status = ui.column().classes("w-full gap-1")

    self.level.on_value_change(lambda: self._fill_networks())
    for facet in (self.category, self.scenario, self.switches):
        facet.on_value_change(lambda: self._fill_networks())

    async def show(self, source: NetworkRef | None, bus: str | None = None) -> None:
    """Load and show the elements of ``source`` (all, or those on ``bus``)."""
    self.source, self.bus = source, bus
    self.label.set_text(f"Bus {bus}" if bus else "All buses")
    self.clear_button.set_visibility(bus is not None)
    if source is None:
        self.box.clear()
        return
    busy(self.box, "Reading elements …")
    try:
        rows = await run.io_bound(self.services.grid.elements, source, bus)
    except Exception as e:  # noqa: BLE001  show the error on the page
        failed(self.box, e)
        return
    self.box.clear()
    with self.box:
        if not rows:
            ui.label("No loads, generators or storage.").classes("mls-card-caption")
            return
        data_table([asdict(r) for r in rows], ELEMENT_COLUMNS, pagination=10)

    ProfileInspector ¶

    ProfileInspector(
    services: Services, catalogue: bool = False
)

    The network's SimBench profiles: a selectable table and two plots.

    The first plot shows the profiles as SimBench stores them, factors of an element's rated power (1.0 = rated power). The second shows them scaled by the rated power of the network's elements that use them, in kW: the power they draw or feed in on the unmodified network.

    With catalogue=True it shows SimBench's complete profile catalogue instead of one network's profiles, without the scaled plot (there are no elements), and above the table an overview of the profile classes that filters it.

    async def show(self, source: NetworkRef | None = None) -> None:
    """Load the profiles (of ``source``, or the catalogue); plot the first two."""
    self.source = source
    busy(self.table_box, "Reading profiles …")
    self.plot_box.clear()
    try:
        if self.catalogue:
            types = await run.io_bound(self.services.profiles.catalogue_types)
        else:
            types = await run.io_bound(self.services.profiles.profile_types, source)
    except Exception as e:  # noqa: BLE001
        failed(self.table_box, e)
        return
    if self.source != source:  # another network was picked meanwhile
        return
    self.loaded = True
    if self.overview:
        self.overview.show(types)
    self.table_box.clear()
    with self.table_box:
        if not types:
            ui.label(
                "This network ships no profiles (pandapower built-in networks "
                "have none; SimBench networks do)."
            ).classes("mls-card-caption")
            self.selected = []
            return
        rows = profile_rows(types)
        if self.catalogue:  # no network, so no element counts
            for row in rows:
                row.pop("elements")
        search = (
            ui.input(placeholder="Search, e.g. H0, PV, wind, power_plant")
            .props("dense clearable")
            .classes("w-full")
        )
        table = (
            ui.table(
                columns=[
                    {
                        "name": k,
                        "label": k.capitalize(),
                        "field": k,
                        "align": "left",
                        "sortable": True,
                    }
                    for k in rows[0]
                ],
                rows=rows,
                row_key="code",
                selection="multiple",
                pagination=10,
            )
            .classes("w-full")
            .props("flat dense")
        )
        search.bind_value(table, "filter")
        self.rows, self.table = rows, table
        table.selected = rows[:2]
        table.on_select(lambda e: self.plot([r["code"] for r in e.selection]))
    await self.plot([r["code"] for r in rows[:2]])

    async def plot(self, codes: list[str] | None = None) -> None:
    """Plot the selected profiles over the chosen window and resolution."""
    if codes is not None:
        self.selected = codes
    if not self.loaded:
        return
    if not self.selected:
        for box in (self.plot_box, self.power_box):
            box.clear()
            with box:
                ui.label("Select profiles in the table.").classes(
                    "mls-card-caption"
                )
        return
    window = (
        self.selected,
        self.day.value,
        int(self.span.value),
        self.resolution.value,
        self.quantity.value,
    )
    if self.catalogue:
        await self._draw(
            self.plot_box, self.services.profiles.catalogue_series, window
        )

    window = (self.source, *window)
    await self._draw(self.plot_box, self.services.profiles.profile_series, window)
    await self._draw(
        self.power_box,
        self.services.profiles.profile_power,
        (*window, bool(self.per_element.value)),
    )
